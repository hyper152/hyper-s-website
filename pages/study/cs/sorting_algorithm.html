void bubbleSort(int a[], int n) { for (int i = 0; i < n-1; ++i) for (int j = 0; j < n-i-1; ++j) if (a[j] > a[j+1]) swap(a[j], a[j+1]); }

void selectionSort(int a[], int n) { for (int i = 0; i < n-1; ++i) { int minIdx = i; for (int j = i+1; j < n; ++j) if (a[j] < a[minIdx]) minIdx = j; swap(a[i], a[minIdx]); } }

void insertionSort(int a[], int n) { for (int i = 1; i < n; ++i) { int key = a[i], j = i-1; while (j >= 0 && a[j] > key) { a[j+1] = a[j]; --j; } a[j+1] = key; } }
 将元素插入已排序区
希尔 O(n log n)~O(n²) ❌ 不稳定
void shellSort(int a[], int n) { for (int gap = n/2; gap > 0; gap /= 2) for (int i = gap; i < n; ++i) { int tmp = a[i], j = i; while (j >= gap && a[j-gap] > tmp) { a[j] = a[j-gap]; j -= gap; } a[j] = tmp; } }

void merge(int a[], int l, int m, int r) { int n1 = m-l+1, n2 = r-m; int L[n1], R[n2]; for (int i=0; i<n1; ++i) L[i]=a[l+i]; for (int j=0; j<n2; ++j) R[j]=a[m+1+j]; int i=0, j=0, k=l; while (i<n1 && j<n2) a[k++] = (L[i] <= R[j]) ? L[i++] : R[j++]; while (i<n1) a[k++] = L[i++]; while (j<n2) a[k++] = R[j++]; } void mergeSort(int a[], int l, int r) { if (l >= r) return; int m = l + (r-l)/2; mergeSort(a,l,m); mergeSort(a,m+1,r); merge(a,l,m,r); }
 分治合并
快速 O(n log n) 平均 ❌ 不稳定
int partition(int a[], int lo, int hi) { int pivot = a[hi], i = lo-1; for (int j = lo; j < hi; ++j) if (a[j] < pivot) swap(a[++i], a[j]); swap(a[i+1], a[hi]); return i+1; } void quickSort(int a[], int lo, int hi) { if (lo < hi) { int p = partition(a, lo, hi); quickSort(a, lo, p-1); quickSort(a, p+1, hi); } }
 分治 + 基准划分
堆 O(n log n) ❌ 不稳定
void heapify(int a[], int n, int i) { int largest = i, l = 2*i+1, r = 2*i+2; if (l < n && a[l] > a[largest]) largest = l; if (r < n && a[r] > a[largest]) largest = r; if (largest != i) { swap(a[i], a[largest]); heapify(a, n, largest); } } void heapSort(int a[], int n) { for (int i = n/2-1; i >= 0; --i) heapify(a, n, i); for (int i = n-1; i > 0; --i) { swap(a[0], a[i]); heapify(a, i, 0); } }
 大根堆 + 交换
计数 O(n + k) ✅ 稳定
void countingSort(int a[], int n, int maxVal) { int count[maxVal+1] = {0}; for (int i=0; i<n; ++i) count[a[i]]++; int idx = 0; for (int i=0; i<=maxVal; ++i) while (count[i]--) a[idx++] = i; }
 非比较，统计频率

void bucketSort(float a[], int n) { vector<vector<float>> buckets(n); for (int i=0; i<n; ++i) buckets[(int)(n * a[i])].push_back(a[i]); for (auto& b : buckets) sort(b.begin(), b.end()); int idx = 0; for (auto& b : buckets) for (float v : b) a[idx++] = v; }
 分散到桶，桶内稳定则稳定
基数 O(d·(n+k)) ✅ 稳定
void radixSort(int a[], int n) { int maxVal = *max_element(a, a+n); for (int exp = 1; maxVal/exp > 0; exp *= 10) { int output[n], count[10] = {0}; for (int i=0; i<n; ++i) count[(a[i]/exp)%10]++; for (int i=1; i<10; ++i) count[i] += count[i-1]; for (int i=n-1; i>=0; --i) { output[count[(a[i]/exp)%10] - 1] = a[i]; count[(a[i]/exp)%10]--; } for (int i=0; i<n; ++i) a[i] = output[i]; } }
