冒泡排序

O(n²)
相邻元素两两比较，大的逐渐“浮”到末尾。
void bubbleSort(int arr[], int n) {
        for (int i = 0; i < n - 1; ++i)
            for (int j = 0; j < n - i - 1; ++j)
                if (arr[j] > arr[j + 1])
                    swap(arr[j], arr[j + 1]);
    }
稳定 相邻交换 · 不改变相等元素相对顺序

选择排序

O(n²)
每轮选出最小元素，与当前边界交换。
void selectionSort(int arr[], int n) {
        for (int i = 0; i < n - 1; ++i) {
            int minIdx = i;
            for (int j = i + 1; j < n; ++j)
                if (arr[j] < arr[minIdx]) minIdx = j;
            swap(arr[i], arr[minIdx]);
        }
    }
不稳定 跳跃交换 · 可能破坏相等元素顺序

插入排序

O(n²)
将元素插入已排序部分的正确位置。
void insertionSort(int arr[], int n) {
        for (int i = 1; i < n; ++i) {
            int key = arr[i], j = i - 1;
            while (j >= 0 && arr[j] > key) {
                arr[j + 1] = arr[j];
                --j;
            }
            arr[j + 1] = key;
        }
    }
稳定 局部移动 · 相等元素不交叉

希尔排序

O(n log n) ~ O(n²)
插入排序的改进版，按增量分组排序。
void shellSort(int arr[], int n) {
        for (int gap = n / 2; gap > 0; gap /= 2)
            for (int i = gap; i < n; ++i) {
                int tmp = arr[i], j = i;
                while (j >= gap && arr[j - gap] > tmp) {
                    arr[j] = arr[j - gap];
                    j -= gap;
                }
                arr[j] = tmp;
            }
    }
不稳定 分组跳跃 · 相同元素可能被分到不同组

归并排序

O(n log n)
分治思想，先递归拆分，再有序合并。
void merge(int arr[], int l, int m, int r) {
        int n1 = m - l + 1, n2 = r - m;
        int L[n1], R[n2];
        for (int i = 0; i < n1; ++i) L[i] = arr[l + i];
        for (int j = 0; j < n2; ++j) R[j] = arr[m + 1 + j];
        int i = 0, j = 0, k = l;
        while (i < n1 && j < n2)
            arr[k++] = (L[i] <= R[j]) ? L[i++] : R[j++];
        while (i < n1) arr[k++] = L[i++];
        while (j < n2) arr[k++] = R[j++];
    }
    void mergeSort(int arr[], int l, int r) {
        if (l >= r) return;
        int m = l + (r - l) / 2;
        mergeSort(arr, l, m);
        mergeSort(arr, m + 1, r);
        merge(arr, l, m, r);
    }
稳定 合并时 ≤ 保证相等元素顺序

快速排序

O(n log n) 平均
选基准 pivot，分割成左右两部分递归。
int partition(int arr[], int low, int high) {
        int pivot = arr[high];
        int i = low - 1;
        for (int j = low; j < high; ++j)
            if (arr[j] < pivot) swap(arr[++i], arr[j]);
        swap(arr[i + 1], arr[high]);
        return i + 1;
    }
    void quickSort(int arr[], int low, int high) {
        if (low < high) {
            int pi = partition(arr, low, high);
            quickSort(arr, low, pi - 1);
            quickSort(arr, pi + 1, high);
        }
    }
不稳定 分区时跳跃交换 · 相等元素可能乱序

堆排序

O(n log n)
利用大根堆反复取最大值，就地排序。
void heapify(int arr[], int n, int i) {
        int largest = i, l = 2*i + 1, r = 2*i + 2;
        if (l < n && arr[l] > arr[largest]) largest = l;
        if (r < n && arr[r] > arr[largest]) largest = r;
        if (largest != i) {
            swap(arr[i], arr[largest]);
            heapify(arr, n, largest);
        }
    }
    void heapSort(int arr[], int n) {
        for (int i = n/2 - 1; i >= 0; --i) heapify(arr, n, i);
        for (int i = n - 1; i > 0; --i) {
            swap(arr[0], arr[i]);
            heapify(arr, i, 0);
        }
    }
不稳定 堆调整时跳跃交换 · 相等元素无法保证顺序

计数排序

O(n + k)
非比较排序，统计元素出现次数并回填。
void countingSort(int arr[], int n, int maxVal) {
        int count[maxVal + 1] = {0};
        for (int i = 0; i < n; ++i) count[arr[i]]++;
        int idx = 0;
        for (int i = 0; i <= maxVal; ++i)
            while (count[i]--) arr[idx++] = i;
    }
稳定 按顺序回填 · 相等元素保持原序（稳定版需额外数组）

桶排序

O(n + k) 平均
将元素分散到有序桶中，桶内再排序。
void bucketSort(float arr[], int n) {
        // 假设元素在 [0,1) 区间
        vector<vector<float>> buckets(n);
        for (int i = 0; i < n; ++i)
            buckets[int(n * arr[i])].push_back(arr[i]);
        for (auto &b : buckets) sort(b.begin(), b.end());
        int idx = 0;
        for (auto &b : buckets)
            for (float v : b) arr[idx++] = v;
    }
稳定 若桶内用稳定排序（如插入排序）则整体稳定

基数排序

O(d · (n + k))
按位（个位、十位……）多次分配收集。
void radixSort(int arr[], int n) {
        int maxVal = *max_element(arr, arr + n);
        for (int exp = 1; maxVal / exp > 0; exp *= 10) {
            int output[n], count[10] = {0};
            for (int i = 0; i < n; ++i) count[(arr[i]/exp)%10]++;
            for (int i = 1; i < 10; ++i) count[i] += count[i-1];
            for (int i = n - 1; i >= 0; --i) {
                output[count[(arr[i]/exp)%10] - 1] = arr[i];
                count[(arr[i]/exp)%10]--;
            }
            for (int i = 0; i < n; ++i) arr[i] = output[i];
        }
    }
稳定 按位分配时使用稳定计数排序
